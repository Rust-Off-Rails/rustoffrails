//! `app task api_token <email> [name] [--read-only] [--days N]`: mints an API token for an
//! account and prints it once (operators, CI, a first client). Written by `generate api`.

use chrono::Utc;

use crate::models::{api_token, user};
use crate::state::AppState;

pub async fn run(state: &AppState, args: Vec<String>) -> anyhow::Result<()> {
    let mut positional = Vec::new();
    let mut write = true;
    let mut days: Option<i64> = None;
    let mut it = args.into_iter();
    while let Some(a) = it.next() {
        match a.as_str() {
            "--read-only" => write = false,
            "--days" => {
                let n = it.next().and_then(|d| d.parse::<i64>().ok());
                days = Some(n.filter(|d| (1..=365).contains(d)).ok_or_else(|| {
                    anyhow::anyhow!("--days takes 1 to 365")
                })?);
            }
            _ => positional.push(a),
        }
    }
    let Some(email) = positional.first() else {
        anyhow::bail!("usage: app task api_token <email> [name] [--read-only] [--days N]");
    };
    let name = positional.get(1).map_or("task", String::as_str);
    let Some(person) = user::find_by_email(&state.db, email).await? else {
        anyhow::bail!("no account for {email}");
    };
    let now = Utc::now();
    let expires = days.map(|d| now + chrono::Duration::days(d));
    let (row, raw) = api_token::issue(&state.db, &person, name, write, expires, now).await?;
    // The only time the token is shown: it is stored as a digest.
    println!("token {} for {email} ({}): {raw}", row.id, row.scope_list().join(","));
    Ok(())
}
